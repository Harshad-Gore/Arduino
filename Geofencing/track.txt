from serial.tools import list_ports
import folium
from datetime import datetime
import time
import webbrowser
import os

# Configuration
HOME_LAT = 18.67032344
HOME_LON = 73.89045092
SAFE_RADIUS = 130.0
MAP_FILE = "arrest_monitor.html"

PERSON_DETAILS = {
    "name": "Kunal Garud",
    "age": 20,
    "case_number": "HR/2024/7890",
    "offense": "Bail Violation",
    "allowed_radius": f"{SAFE_RADIUS} meters",
    "tracking_id": "GPS-ALERT-001",
    "officer_in_charge": "Srikant Tiwari",
    "last_seen": "Initializing..."
}

HOME_DETAILS = {
    "address": "12B, Baker Street",
    "landmark": "Near City Hospital",
    "city": "Mumbai",
    "jurisdiction": "Mumbai Police Station No.5"
}

def get_arduino_port():
    """Find and return the Arduino COM port"""
    ports = list_ports.comports()
    for p in ports:
        if 'Arduino' in p.description or 'CH340' in p.description:  # CH340 for clones
            return p.device
    return None

def create_map():

    # Home Marker
    folium.Marker(

        tooltip="Restricted Residence",

        popup=f"""<b>Residence Details:</b><br>
        {HOME_DETAILS['address']}<br>

        <h3 style="color: #cc0000; margin-top:0;">🚨 Home Arrest Monitor</h3>

    m.get_root().html.add_child(folium.Element(info_html))

def update_map(m, lat, lng, distance):
    # Current Position Marker
    folium.Marker(
        [lat, lng],
        tooltip=f"""<b>Current Position</b><br>
        {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}<br>
        Distance: {distance:.1f}m""",
        icon=folium.Icon(color='blue' if distance <= SAFE_RADIUS else 'red', icon='user')
    ).add_to(m)
    
    # Geofence Circle
    folium.Circle(
        location=[HOME_LAT, HOME_LON],
        radius=SAFE_RADIUS,
        color='red',
        fill=True,
        fill_color='red',
        fill_opacity=0.2
    ).add_to(m)
    
    # Connection Line
    folium.PolyLine(
        locations=[[HOME_LAT, HOME_LON], [lat, lng]],
        color='blue',
        weight=2,
        dash_array='5'
    ).add_to(m)
    
    # Update timestamp
    m.get_root().html.add_child(folium.Element(f"""
    <script>
    document.getElementById('updateTime').innerHTML = '{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}';
    </script>
    """))
    
    m.save(MAP_FILE)

def main():
    # Setup serial connection
    arduino_port = get_arduino_port()
    if not arduino_port:
        print("Arduino not found! Check USB connection.")
        return

    try:
        ser = serial.Serial(arduino_port, 9600, timeout=1)
    except Exception as e:
        print(f"Connection failed: {str(e)}")
        return

    print(f"Connected to {arduino_port}")
    
    # Create initial map
    m = create_map()
    m.save(MAP_FILE)
    webbrowser.open(f"file://{os.path.abspath(MAP_FILE)}")
    print("Map opened in browser. Refresh to see updates!")

    while True:
        try:
            line = ser.readline().decode().strip()
            
            if line.startswith("TRACK:"):
                data = line.split(":")[1].split(",")
                lat = float(data[0])
                lng = float(data[1])
                distance = float(data[2])
                timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

                update_map(m, lat, lng, distance)
                
                status = "WITHIN LIMITS" if distance <= SAFE_RADIUS else "ALERT: OUT OF BOUNDS!"
                print(f"""
[{timestamp}] {status}
Coordinates: {lat:.6f}, {lng:.6f}
Distance: {distance:.1f}m
Case: {PERSON_DETAILS['case_number']}
Officer: {PERSON_DETAILS['officer_in_charge']}
""")

            elif line.startswith("STATUS:"):
                print(line)

        except ValueError as ve:
            print(f"Invalid data format: {str(ve)}")
        except IndexError as ie:
            print(f"Data parsing error: {str(ie)}")
        except Exception as e:
            print(f"General error: {str(e)}")
            time.sleep(1)

if __name__ == "__main__":
    main()
